Conceitos: LIFO, notação pós-fixa
Problema: Guardar itens de modo que o último a entrar seja o primeiro a sair (LIFO, last in, first out), e usar essa disciplina para calcular expressões em notação pós-fixa.

Ideia. Uma pilha oferece duas operações, ambas no mesmo extremo, o topo: inserir (push) coloca um item no topo; remover (pop) retira e devolve o item do topo. Como numa pilha de pratos, só se alcança o prato de cima. A pilha pode ser implementada com um vetor e um índice topo (espaço limitado por um tamanho máximo) ou com uma lista ligada (espaço limitado apenas pela memória; veja o tópico "Lista ligada simples" para os detalhes de nós e apontadores).

Invariante (vetor). Os itens da pilha são pilha[0..topo], com pilha[topo] sendo o topo; o índice topo começa em -1, e a pilha está vazia quando topo == -1 e cheia quando topo == MAX-1. inserir incrementa topo e depois escreve em pilha[topo]; remover lê pilha[topo] e depois decrementa topo. Inserir em pilha cheia e remover de pilha vazia são erros e devem ser tratados: as funções devolvem um código de erro.

Pilha em lista ligada. O código do segundo trecho usa um nó cabeça (sentinela) cujo prox aponta para o topo; inserir e remover acontecem logo depois do cabeça, em O(1), sem nenhum caso especial para a pilha vazia. (Também é possível inserir e remover no último nó da lista, mas isso custa O(n) por operação, pois exige percorrer a lista toda.)

Complexidade. inserir, remover e consulta do topo custam O(1), nas duas implementações (na lista ligada, com a inserção no início).

Notação pós-fixa. Na notação infixa escrevemos 3 + 4 e precisamos de precedências e parênteses para decidir a ordem das operações. Na pós-fixa, o operador vem depois dos operandos: 3 4 +. A expressão infixa 2 * (3 + 1) vira 2 3 1 + *, sem parênteses. Para calcular uma expressão pós-fixa, varre-se da esquerda para a direita: um operando é empilhado; um operador desempilha dois operandos (o primeiro a sair é o operando da direita), aplica a operação e empilha o resultado. Ao fim, a pilha tem um único valor, o resultado. O primeiro trecho de código faz isso para operandos de um dígito.

<<C
#include <stdio.h>
#include <ctype.h>

#define MAX 10

int pilha[MAX];
int topo = -1;                       /* pilha vazia */

void lista_pilha(void)
{
  int i;
  printf("topo=%d: ", topo);
  for (i = 0; i <= topo; i++)
    printf("%d ", pilha[i]);
  printf("\n");
}

/* Insere no topo. Devolve a posição usada, ou -1 se a pilha está cheia. */
int inserir(int x)
{
  if (topo == MAX - 1) {
    printf("Erro: pilha cheia (%d)\n", MAX);
    return -1;
  }
  pilha[++topo] = x;
  return topo;
}

/* Remove do topo. Devolve o item, ou -2 se a pilha está vazia. */
int remover(void)
{
  if (topo < 0) {
    printf("Erro: pilha vazia\n");
    return -2;
  }
  return pilha[topo--];
}

/* Avalia uma expressão pós-fixa com operandos de 1 dígito e + - * /.
   Devolve 0 se ok (resultado em *res), -1 se a expressão é inválida. */
int avaliaPosfixa(const char *s, int *res)
{
  int a, b;

  topo = -1;
  for (; *s != '\0'; s++) {
    if (isspace((unsigned char)*s))
      continue;
    if (isdigit((unsigned char)*s)) {
      if (inserir(*s - '0') < 0)
        return -1;
    } else {
      if (topo < 1)
        return -1;                   /* faltam operandos */
      b = remover();                 /* o operando da direita sai primeiro */
      a = remover();
      switch (*s) {
      case '+': inserir(a + b); break;
      case '-': inserir(a - b); break;
      case '*': inserir(a * b); break;
      case '/':
        if (b == 0)
          return -1;
        inserir(a / b);
        break;
      default:
        return -1;
      }
    }
  }
  if (topo != 0)
    return -1;                       /* deve sobrar exatamente um valor */
  *res = remover();
  return 0;
}

int main(void)
{
  int r;

  inserir(1); inserir(2); inserir(3);
  remover();
  lista_pilha();                     /* topo=1: 1 2 */
  topo = -1;

  if (avaliaPosfixa("2 3 1 + *", &r) == 0)
    printf("%d\n", r);               /* 2 * (3 + 1) = 8 */
  if (avaliaPosfixa("9 3 / 2 -", &r) == 0)
    printf("%d\n", r);               /* 9 / 3 - 2 = 1 */
  printf("%d\n", avaliaPosfixa("3 +", &r));   /* -1: faltou operando */
  return 0;
}

<<C
#include <stdio.h>
#include <stdlib.h>

typedef struct NoTemp {
  int info;
  struct NoTemp *prox;
} No;

/* Cria um nó com informação i. */
No *criarNo(int i)
{
  No *no = malloc(sizeof(No));
  if (no == NULL)
    exit(EXIT_FAILURE);
  no->info = i;
  no->prox = NULL;
  return no;
}

/* Empilha: insere o nó logo depois do cabeça (que é o topo). */
void empilha(No *cab, No *no)
{
  no->prox = cab->prox;
  cab->prox = no;
}

/* Desempilha: retira o nó do topo e o devolve (NULL se a pilha está vazia).
   Quem chama deve liberar o nó com free. */
No *desempilha(No *cab)
{
  No *no = cab->prox;
  if (no != NULL)
    cab->prox = no->prox;
  return no;
}

void lista(No *cab)
{
  No *aux;
  printf("Pilha (topo à esquerda): ");
  for (aux = cab->prox; aux != NULL; aux = aux->prox)
    printf("(%2d) ", aux->info);
  printf("\n");
}

int main(void)
{
  No *cabeca = criarNo(-1);          /* nó cabeça: não guarda dado */
  No *aux;
  int i;

  aux = desempilha(cabeca);          /* pilha vazia: NULL */
  printf("%s\n", aux == NULL ? "vazia" : "erro");
  for (i = 0; i < 5; i++)
    empilha(cabeca, criarNo(i));
  lista(cabeca);                     /* 4 3 2 1 0 */
  aux = desempilha(cabeca);
  printf("Removido %d\n", aux->info);          /* 4 */
  free(aux);
  empilha(cabeca, criarNo(6));
  lista(cabeca);                     /* 6 3 2 1 0 */
  while ((aux = desempilha(cabeca)) != NULL)
    free(aux);
  free(cabeca);
  return 0;
}
