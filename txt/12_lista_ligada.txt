Tecnica: Estrutura dinâmica
Algoritmo: Lista ligada simples (insere fim, remove por id/nome)
Autores:
Data:
Conceitos: alocação dinâmica, union, nó cabeça, lista ligada
Problema: Manter uma coleção de registros cujo tamanho varia durante a execução, com inserção no fim e remoção por identificador ou por nome, sem reservar espaço de antemão.

Ideia. Uma lista ligada é uma sequência de nós alocados dinamicamente (malloc); cada nó guarda um registro e um apontador prox para o nó seguinte, e o último nó tem prox == NULL, marcando o fim da lista. Não há índices: para chegar ao k-ésimo nó é preciso seguir k apontadores. Em troca, inserir e remover não obrigam a deslocar elementos, como num vetor. Quando o registro é criado com malloc, o espaço continua existindo depois que a função que o criou termina, e só deixa de existir com free; por isso, depois de usar um registro que não é mais necessário, devolva-o ao "banco de memória" com free.

Construção. criaRegistro aloca um registro e devolve um apontador para ele, com prox == NULL. Para formar a lista, o apontador primeiro guarda o início; inserir no fim é fazer o prox do último nó apontar para o novo registro.

Casos extremos. Nas remoções (por nome ou por id), cuidado com as situações extremas: lista vazia (nada a fazer), item a remover é o primeiro (o início da lista muda: por isso as funções devolvem o novo início), item no meio ou item que é o último (o anterior passa a apontar para NULL). Para isso mantemos dois apontadores durante a busca, ant (o anterior) e p (o atual), com ant->prox == p; remover p é fazer ant->prox = p->prox e free(p). O mesmo vale para a inserção no fim: se a lista está vazia, o novo registro passa a ser o primeiro.

Nó cabeça. Uma alternativa que elimina esses casos especiais é o nó cabeça (sentinela): um nó fixo, sem dado útil, que é sempre o primeiro. A lista vazia é o cabeça com prox == NULL; assim, inserir e remover no início ou no meio seguem a mesma regra, e o início da lista nunca muda. Os algoritmos [[pilha]] e [[fila]] usam essa técnica.

Invariante. Em qualquer momento, a partir de primeiro os nós formam uma cadeia sem ciclos terminada em NULL; cada nó foi obtido com malloc e é liberado com free exatamente uma vez.

Complexidade. Inserir no fim exige percorrer a lista: O(n) (é O(1) se se guarda também um apontador para o último nó). Remover por id ou por nome exige uma busca: O(n). O espaço é O(n) mais um apontador por nó.

União (union). Uma union guarda, no mesmo espaço, um valor de um entre vários tipos; o programa precisa lembrar qual é o tipo ativo, por isso costuma vir acompanhada de uma etiqueta (campo tipo). O segundo trecho de código mostra uma lista cujos nós guardam um inteiro ou um número em ponto flutuante, conforme o campo tipo ('i' ou 'f').

Operações adicionais. O terceiro trecho resolve operações clássicas sobre listas: contar os nós, buscar um nó e inverter a lista no lugar.

<<C
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define TAM 10                       /* limite para o campo nome */

typedef struct reg {
  int id;
  char nome[TAM];
  struct reg *prox;                  /* aponta para o próximo registro */
} tipoRegistro;

/* Cria um registro e devolve um apontador para ele (NULL se faltar memória). */
tipoRegistro *criaRegistro(char *nome, int id)
{
  tipoRegistro *item = malloc(sizeof(tipoRegistro));
  if (item == NULL)
    return NULL;
  strncpy(item->nome, nome, TAM - 1);
  item->nome[TAM - 1] = '\0';
  item->id = id;
  item->prox = NULL;
  return item;
}

/* Exercício 1: insere novo ao fim e devolve o (possivelmente novo) início. */
tipoRegistro *insereFim(tipoRegistro *primeiro, tipoRegistro *novo)
{
  tipoRegistro *p;
  if (primeiro == NULL)              /* lista vazia: novo é o primeiro */
    return novo;
  for (p = primeiro; p->prox != NULL; p = p->prox)
    ;
  p->prox = novo;
  return primeiro;
}

/* Exercício 2: remove o primeiro registro de nome dado; devolve o início. */
tipoRegistro *removePorNome(tipoRegistro *primeiro, char *nome)
{
  tipoRegistro *ant = NULL, *p = primeiro;
  while (p != NULL && strcmp(p->nome, nome) != 0) {
    ant = p;
    p = p->prox;
  }
  if (p == NULL)
    return primeiro;                 /* lista vazia ou nome inexistente */
  if (ant == NULL)
    primeiro = p->prox;              /* era o primeiro */
  else
    ant->prox = p->prox;             /* estava no meio ou era o último */
  free(p);
  return primeiro;
}

/* Exercício 3: remove o primeiro registro de id dado; devolve o início. */
tipoRegistro *removePorId(tipoRegistro *primeiro, int id)
{
  tipoRegistro *ant = NULL, *p = primeiro;
  while (p != NULL && p->id != id) {
    ant = p;
    p = p->prox;
  }
  if (p == NULL)
    return primeiro;
  if (ant == NULL)
    primeiro = p->prox;
  else
    ant->prox = p->prox;
  free(p);
  return primeiro;
}

void imprimeLista(tipoRegistro *p)
{
  for (; p != NULL; p = p->prox)
    printf(" - %d : %s\n", p->id, p->nome);
  printf("\n");
}

void liberaLista(tipoRegistro *p)
{
  while (p != NULL) {
    tipoRegistro *prox = p->prox;
    free(p);
    p = prox;
  }
}

int main(void)
{
  int i;
  char nomes[][TAM] = { "aluno 1", "aluno 2", "aluno 3", "aluno 4", "aluno 5" };
  tipoRegistro *primeiro = NULL;

  for (i = 0; i < 5; i++)
    primeiro = insereFim(primeiro, criaRegistro(nomes[i], i + 1));
  imprimeLista(primeiro);
  primeiro = removePorNome(primeiro, "aluno 1");   /* o primeiro */
  primeiro = removePorId(primeiro, 5);             /* o último */
  primeiro = removePorId(primeiro, 3);             /* um do meio */
  primeiro = removePorNome(primeiro, "ninguem");   /* não existe */
  imprimeLista(primeiro);                          /* 2 e 4 */
  liberaLista(primeiro);
  primeiro = removePorId(NULL, 1);                 /* lista vazia: continua NULL */
  return 0;
}

#include <stdio.h>
#include <stdlib.h>

typedef struct TipoNo {
  char tipo;                         /* 'i' => inteiro ; 'f' => ponto flutuante */
  union { float f; int i; } info;    /* ocupa o espaço de um só deles */
  struct TipoNo *prox;
} TipoNo;

/* Cria um nó do tipo pedido; usa vali se tipo == 'i' e valf caso contrário. */
TipoNo *adiciona_no(char tipo, float valf, int vali)
{
  TipoNo *aux = malloc(sizeof(TipoNo));
  if (aux == NULL)
    exit(EXIT_FAILURE);
  aux->tipo = tipo;
  if (tipo == 'i')
    aux->info.i = vali;
  else
    aux->info.f = valf;
  aux->prox = NULL;
  return aux;
}

void imprime_lista(TipoNo *r)
{
  for (; r != NULL; r = r->prox) {
    if (r->tipo == 'i')
      printf("%d\n", r->info.i);
    else
      printf("%f\n", r->info.f);
  }
}

int main(void)
{
  /* [ i; 1 ] -> [ f; 2.0 ] -> [ f; 3.4456 ] -> [ i; 4 ] */
  TipoNo *no = adiciona_no('i', 0, 1), *p;

  no->prox = adiciona_no('f', 2.0f, 0);
  no->prox->prox = adiciona_no('f', 3.4456f, 0);
  no->prox->prox->prox = adiciona_no('i', 0, 4);
  imprime_lista(no);
  while (no != NULL) {
    p = no->prox;
    free(no);
    no = p;
  }
  return 0;
}

<<C
#include <stdio.h>
#include <stdlib.h>

typedef struct cel {
    int info;
    struct cel *prox;
} Cel;

/* Insere no início e devolve o novo início. */
static Cel *insereInicio(Cel *ini, int x)
{
    Cel *p = malloc(sizeof *p);
    if (p == NULL)
        exit(EXIT_FAILURE);
    p->info = x;
    p->prox = ini;
    return p;
}

/* Número de nós da lista. */
int comprimento(const Cel *ini)
{
    int n = 0;
    for (; ini != NULL; ini = ini->prox)
        n++;
    return n;
}

/* Procura x; devolve o nó ou NULL. */
Cel *buscaLista(Cel *ini, int x)
{
    while (ini != NULL && ini->info != x)
        ini = ini->prox;
    return ini;
}

/* Inverte a lista no lugar (sem alocar nós). Devolve o novo início.
   Invariante: "ant" é a parte já invertida; "ini" é o que falta inverter. */
Cel *inverte(Cel *ini)
{
    Cel *ant = NULL;
    while (ini != NULL) {
        Cel *prox = ini->prox;
        ini->prox = ant;
        ant = ini;
        ini = prox;
    }
    return ant;
}

int main(void)
{
    Cel *l = NULL;
    for (int i = 1; i <= 4; i++)
        l = insereInicio(l, i);               /* 4 3 2 1 */
    printf("%d\n", comprimento(l));      /* 4 */
    printf("%d\n", buscaLista(l, 3) != NULL);   /* 1 */
    l = inverte(l);
    for (const Cel *p = l; p != NULL; p = p->prox)
        printf("%d ", p->info);          /* 1 2 3 4 */
    printf("\n");
    while (l != NULL) {
        Cel *p = l->prox;
        free(l);
        l = p;
    }
    return 0;
}
